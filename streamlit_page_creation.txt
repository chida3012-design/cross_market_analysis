def Filters_and_Data_Exploration():
        st.header(" :material/data_exploration: Filters & Data Exploration",divider="grey")
        tab1, tab2 = st.tabs([" :material/currency_rupee: Avg Price", " :material/table_chart_view: Daily market"])
        with tab1:
            st.subheader("select a date range to know the average price of various financial assets.")
            with st.container(horizontal=True):
                with st.popover("Start date"):
                    st.markdown("Start date")
                    d1= st.date_input("Start date", value=None,width="stretch")
                st.write(" Start date", d1)
                st.space("stretch")
                with st.popover("End date"):
                    st.markdown("End date")
                    d2= st.date_input("End date", value=None,width="stretch")
                st.write(" End date", d2)



            if d1 and d2 :
              option1a_df =pd.read_sql_query("""select avg(price) as avg_bitcoin_price
              from cryptoprices where coin_id='bitcoin' and (date between '{}'and'{}');""".format(d1,d2),conn)
              bit=round(option1a_df.iat[0,0],2)

              option1b_df =pd.read_sql_query("""select avg(price_inr) as avg_oil_price
              from oil_prices where  (date between '{}'and'{}');""".format(d1,d2),conn)
              oil=round(option1b_df.iat[0,0],2)

              option1c_df =pd.read_sql_query("""select avg(close) as avg_SP_price
              from stock_prices where ticker='GSPC' and  (date between '{}'and'{}');""".format(d1,d2),conn)
              gspc=round(option1c_df.iat[0,0],2)

              option1d_df =pd.read_sql_query("""select avg(close) as avg_NIFTY_price
              from stock_prices where ticker='NSEI' and  (date between '{}'and'{}');""".format(d1,d2),conn)
              nsei=round(option1c_df.iat[0,0],2)

              a, b = st.columns(2)
              c, d = st.columns(2)

              a.metric(" :material/currency_bitcoin: avg_bitcoin_price",value=bit, border=True)
              b.metric(":material/oil_barrel: avg_oil_price",value=oil, border=True)

              c.metric(" :material/currency_exchange: avg_S&P_price",value=gspc, border=True)
              d.metric(" :material/currency_exchange: avg_NIFTY_price",value=nsei, border=True)


        with tab2:
                st.subheader(" :material/table: Daily market snapshot table")
                option1_df = pd.read_sql_query("""SELECT
                        c.date,
                        o.price_inr as oil_price,
                        SUM(CASE WHEN c.coin_id='bitcoin' THEN c.price ELSE 0 END) AS bitcoin_price,
                        SUM(CASE WHEN s.ticker='NSEI' THEN s.close ELSE 0 END) AS NSEI_close,
                        SUM(CASE WHEN s.ticker='GSPC' THEN s.close ELSE 0 END) AS GSPC_close
                    FROM
                        cryptoprices c
                    join
                        oil_prices o on o.date = c.date
                    JOIN
                        stock_prices s ON s.date = c.date
                    where
                        s.ticker IN ('NSEI', 'GSPC')
                        GROUP BY c.date,o.price_inr
                        order by c.date desc;""",conn)
                st.dataframe(option1_df)
                st.line_chart(
                option1_df,
                x="date",
                y=["bitcoin_price", "oil_price","NSEI_close","GSPC_close"],
                color=["red", "blue","green","yellow"],
            )

                      / MAX(CASE WHEN date = '2025-02-21' THEN price END)) * 100 AS percentage_change

      st.dataframe(query14_df)

    elif option == "15.Calculate the volatility of oil prices (max-min difference per year)":
      query15_df = pd.read_sql_query("""select year(date),(max(price_inr)-min(price_inr)) as volatility from oil_prices
      group by year(date);""",conn)
      st.dataframe(query15_df)

    elif option == "16.Get all stock prices for a given ticker":
      query16_df = pd.read_sql_query("""select * from stock_prices
      where ticker='NSEI';""",conn)
      st.dataframe(query16_df)

    elif option == "17.Find the highest closing price for NASDAQ (^IXIC)":
      query17_df = pd.read_sql_query("""select * from stock_prices
      where ticker='IXIC'
      order by close desc limit 1;""",conn)
      st.dataframe(query17_df)

    elif option == "18.List top 5 days with highest price difference (high - low) for S&P 500 (^GSPC)":
      query18_df = pd.read_sql_query("""select ticker,date,(high-low) as difference from stock_prices
      where ticker='GSPC'
      order by difference desc limit 5;""",conn)
      st.dataframe(query18_df)

    elif option == "19.Get monthly average closing price for each ticker":
      query19_df = pd.read_sql_query("""select ticker,month(date),year(date),avg(close) as average from stock_prices
      group by ticker,month(date),year(date)
      order by ticker, year(date) asc,month(date) asc;""",conn)
      st.dataframe(query19_df)

    elif option == "20.Get average trading volume of NSEI in 2024":
      query20_df = pd.read_sql_query("""select ticker,year(date),avg(volume) as average from stock_prices
      where year(date)='2024' and ticker='NSEI'
      group by ticker,year(date);""",conn)
      st.dataframe(query20_df)

    elif option == "21.Compare Bitcoin vs Oil average price in 2025":
      query21_df = pd.read_sql_query("""SELECT
          year(c.date),
          avg(c.price) AS BitcoinAvgPrice,
          avg(o.price_inr) AS OilAvgPrice,
          (avg(c.price) / avg(o.price_inr)) AS BTC_to_Oil_Ratio
      FROM
          cryptoprices c
      JOIN
          oil_prices o ON year(c.date) = year(o.date)
      WHERE
          year(c.date) = '2025' and c.coin_id='bitcoin'
      group by year(c.date),year(o.date);""",conn)
      st.dataframe(query21_df)

    elif option == "22.Check if Bitcoin moves with S&P 500 (correlation idea)":
      query22_df = pd.read_sql_query("""SELECT
          c.date,
          c.price AS BTC_price,
          s.close AS SP_Close,
          ((c.price - LAG(c.price) OVER (ORDER BY c.date)) / LAG(c.price) OVER (ORDER BY c.date)) * 100 AS BTC_Pct_Change,
          ((s.close - LAG(s.close) OVER (ORDER BY s.date)) / LAG(s.close) OVER (ORDER BY s.date)) * 100 AS SP_Pct_Change,
          CASE
              WHEN ((c.price - LAG(c.price) OVER (ORDER BY c.date)) *
                    (s.close - LAG(s.close) OVER (ORDER BY s.date))) > 0
              THEN 'Same Direction'
              ELSE 'Opposite'
          END AS Correlation_Sign
      FROM
          cryptoprices c
      JOIN
          stock_prices s ON c.date = s.date
      WHERE c.coin_id='bitcoin' and s.ticker='GSPC'
      ORDER BY
      c.date DESC;""",conn)
      st.dataframe(query22_df)

    elif option == "23.Compare Ethereum and NASDAQ daily prices for 2025":
      query23_df = pd.read_sql_query("""SELECT
      c.date,
      c.price AS ethereum_price,
      s.close AS IXIC_close_price,
      c.price / s.close AS ethereum_IXIC_Ratio
      FROM
          cryptoprices c
      JOIN
          stock_prices s ON c.date = s.date
      WHERE year(c.date) = '2025' and
      c.coin_id='ethereum' and
      s.ticker='IXIC';""",conn)
      st.dataframe(query23_df)

    elif option == "24.Find days when oil price spiked and compare with Bitcoin price change":
      query24_df = pd.read_sql_query("""SELECT
          o.date,
          o.price_inr AS oil_price,
          ((o.price_inr - prev_o.price_inr) / prev_o.price_inr) * 100 AS oil_percent_change,
          c.price AS btc_price,
          ((c.price - prev_c.price) / prev_c.price) * 100 AS btc_percent_change
      FROM
          oil_prices o
      JOIN
          oil_prices prev_o ON o.date = DATE_ADD(prev_o.date, INTERVAL 1 DAY)
      JOIN
          cryptoprices c ON o.date = c.date
      JOIN
        cryptoprices prev_c ON c.date = DATE_ADD(prev_c.date, INTERVAL 1 DAY)
      WHERE
          c.coin_id='bitcoin' and ((o.price_inr - prev_o.price_inr) / prev_o.price_inr) > 0.02
      ORDER BY
          o.date DESC;""",conn)
      st.dataframe(query24_df)

    elif option == "25.Compare top 3 coins daily price trend vs Nifty (^NSEI)":
      query25_df = pd.read_sql_query("""WITH DailyCryptoChanges AS (
          SELECT
              c.date,
              c.coin_id,
              c.price,
              ((c.price - LAG(c.price) OVER (PARTITION BY c.coin_id ORDER BY c.date))
              / LAG(c.price) OVER (PARTITION BY c.coin_id ORDER BY c.date)) * 100 AS coin_pct_change
          FROM cryptoprices c
          WHERE c.coin_id IN ('bitcoin', 'ethereum', 'tether')
      ),
      DailyNiftyChange AS (
          SELECT
              s.date,
              s.close AS nifty_close,
              ((s.close - LAG(s.close) OVER (ORDER BY s.date))
              / LAG(s.close) OVER (ORDER BY s.date)) * 100 AS nifty_pct_change
          FROM stock_prices s where s.ticker='NSEI'
      )

      SELECT
          d.date,
          d.coin_id,
          ROUND(d.coin_pct_change, 2) AS coin_daily_pct,
          ROUND(n.nifty_pct_change, 2) AS nifty_daily_pct,
          CASE
              WHEN d.coin_pct_change > n.nifty_pct_change THEN 'Outperforming'
              WHEN d.coin_pct_change < n.nifty_pct_change THEN 'Underperforming'
              ELSE 'Neutral'
          END AS trend_comparison
      FROM DailyCryptoChanges d
      JOIN DailyNiftyChange n ON d.date = n.date
      ORDER BY d.date DESC, d.coin_id;""",conn)
      st.dataframe(query25_df)

    elif option == "26.Compare stock prices (^GSPC) with crude oil prices on the same dates":
      query26_df = pd.read_sql_query("""SELECT
            s.date,
            s.close as GSPC_price,
            o.price_inr as oil_price,
            (((s.close - LAG(s.close) OVER (ORDER BY s.date)) / LAG(s.close) OVER (ORDER BY s.date)) *100) AS GSPC_pct_change,
            (((o.price_inr - LAG(o.price_inr) OVER (ORDER BY o.date)) / LAG(o.price_inr) OVER (ORDER BY o.date)) *100) AS oil_pct_change
      from  stock_prices s
      join  oil_prices o on o.date=s.date
      where s.ticker='GSPC'
      order by s.date;;""",conn)
      st.dataframe(query26_df)

    elif option == "27.Correlate Bitcoin closing price with crude oil closing price (same date)":
      query27_df = pd.read_sql_query("""SELECT
          c.date,
          c.price AS BTC_price,
          o.price_inr AS oil_Close,
          ((c.price - LAG(c.price) OVER (ORDER BY c.date)) / LAG(c.price) OVER (ORDER BY c.date)) * 100 AS BTC_Pct_Change,
          ((o.price_inr - LAG(o.price_inr) OVER (ORDER BY o.date)) / LAG(o.price_inr) OVER (ORDER BY o.date)) * 100 AS OIL_Pct_Change,
          CASE
              WHEN ((c.price - LAG(c.price) OVER (ORDER BY c.date)) *
                    (o.price_inr - LAG(o.price_inr) OVER (ORDER BY o.date))) > 0
              THEN 'Same Direction'
              ELSE 'Opposite'
          END AS Correlation_Sign
      FROM
          cryptoprices c
      JOIN
          oil_prices o ON c.date = o.date
      WHERE c.coin_id='bitcoin'
      ORDER BY
          c.date DESC;""",conn)
      st.dataframe(query27_df)

    elif option == "28.Compare NASDAQ (^IXIC) with Ethereum price trends":
      query28_df = pd.read_sql_query("""SELECT
          c.date,
          c.price AS ethereum_price,
          s.close AS IXIC_close_price,
          ((c.price - LAG(c.price) OVER (ORDER BY c.date)) / LAG(c.price) OVER (ORDER BY c.date)) AS Eth_Daily_Return,
          ((s.close - LAG(s.close) OVER (ORDER BY s.date)) / LAG(s.close) OVER (ORDER BY s.date)) AS IXIC_Daily_Return
      FROM
          cryptoprices c
      JOIN
          stock_prices s ON s.date = c.date
      WHERE c.coin_id='ethereum'and s.ticker='IXIC';""",conn)
      st.dataframe(query28_df)

    elif option == "29.Join top 3 crypto coins with stock indices for 2025":
      query29_df = pd.read_sql_query("""SELECT
          c.date,
          SUM(CASE WHEN c.coin_id='bitcoin' THEN c.price ELSE 0 END) AS bitcoin_price,
          SUM(CASE WHEN c.coin_id='ethereum' THEN c.price ELSE 0 END) AS ethereum_price,
          SUM(CASE WHEN c.coin_id='tether' THEN c.price ELSE 0 END) AS tether_price,
          SUM(CASE WHEN s.ticker='NSEI' THEN s.close ELSE 0 END) AS NSEI_close,
          SUM(CASE WHEN s.ticker='GSPC' THEN s.close ELSE 0 END) AS GSPC_close,
          SUM(CASE WHEN s.ticker='IXIC' THEN s.close ELSE 0 END) AS IXIC_close
      FROM
          cryptoprices c
      JOIN
          stock_prices s ON s.date = c.date
      where
          c.coin_id IN ('bitcoin', 'ethereum', 'tether')
          AND s.ticker IN ('NSEI', 'GSPC', 'IXIC')
          and year(c.date)='2025'
          GROUP BY c.date
          order by c.date desc;""",conn)
      st.dataframe(query29_df)

    elif option == "30.Multi-join: stock prices, oil prices, and Bitcoin prices for daily comparison":
      query30_df = pd.read_sql_query("""SELECT
          c.date,
          o.price_inr as oil_price,
          SUM(CASE WHEN c.coin_id='bitcoin' THEN c.price ELSE 0 END) AS bitcoin_price,
          SUM(CASE WHEN c.coin_id='ethereum' THEN c.price ELSE 0 END) AS ethereum_price,
          SUM(CASE WHEN c.coin_id='tether' THEN c.price ELSE 0 END) AS tether_price,
          SUM(CASE WHEN s.ticker='NSEI' THEN s.close ELSE 0 END) AS NSEI_close,
          SUM(CASE WHEN s.ticker='GSPC' THEN s.close ELSE 0 END) AS GSPC_close,
          SUM(CASE WHEN s.ticker='IXIC' THEN s.close ELSE 0 END) AS IXIC_close
      FROM
          cryptoprices c
      join
          oil_prices o on o.date = c.date
      JOIN
          stock_prices s ON s.date = c.date
      where
          c.coin_id IN ('bitcoin', 'ethereum', 'tether')
          AND s.ticker IN ('NSEI', 'GSPC', 'IXIC')
          and c.date >= '2025-01-01'
          GROUP BY c.date,o.price_inr
          order by c.date desc;""",conn)
      st.dataframe(query30_df)
